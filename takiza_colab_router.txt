    def sync_live_models(self):
        """Опрашивает AnyModel API с ключом авторизации для обнаружения свежих моделей."""
        try:
            api_key = os.environ.get("ANYMODEL_API_KEY", "").strip()
            if not api_key:
                return
            headers = {
                "User-Agent": "Takiza-Router/1.0",
                "Authorization": f"Bearer {api_key}"
            }
            req = urllib.request.Request("https://anymodel.org/v1/models", headers=headers)
            with urllib.request.urlopen(req, timeout=4) as resp:
                data = json.loads(resp.read().decode())
                models_list = data.get("data", [])
                for m in models_list:
                    m_id = m.get("id")
                    if m_id and m_id not in self.catalog:
                        # Авто-классификация новой неизвестной модели (например, gpt-5.6-luna)
                        lower = m_id.lower()
                        if any(k in lower for k in ["luna", "astra", "o3", "opus", "max", "pro"]):
                            tier = "frontier_reasoning"
                            in_p, out_p, score = 5.0, 20.0, 97
                        elif any(k in lower for k in ["sonnet", "coder", "deepseek", "qwen"]):
                            tier = "swe_expert"
                            in_p, out_p, score = 2.0, 8.0, 94
                        else:
                            tier = "fast_coder"
                            in_p, out_p, score = 0.3, 1.2, 87
                        self.catalog[m_id] = {
                            "name": m_id,
                            "tier": tier,
                            "in_price": in_p,
                            "out_price": out_p,
                            "coding_score": score,
                            "strengths": ["auto_discovered_online"]
                        }
                print(f"🌐 Синхронизировано {len(self.catalog)} моделей из онлайн-каталога.")
                self.last_sync = time.time()
        except Exception as e:
            print(f"ℹ️ Онлайн каталог AnyModel временно недоступен ({e}), используем встроенный кэш.")

# 6. СЕРВЕР FASTAPI С ПОДДЕРЖКОЙ МЕТРИК
# ==============================================================================
app = FastAPI(title="Takiza MoA Smart Market Router")

class RouteRequest(BaseModel):
    prompt: str

SYSTEM_METRIC_PROMPT = (
    "You are the Takiza MoA Metric Evaluator. Analyze the user's task and classify its technical requirements.\n"
    "Output strictly valid JSON with:\n"
    "1. category: (greeting | routine_backend | frontend_ui | concurrency_debugging | system_architecture)\n"
    "2. complexity_score: float from 0.0 (trivial greeting/chat) to 1.0 (extreme SIMD/distributed/formal logic)\n"
    "3. rationale: brief technical justification of required capability"
)

@app.post("/route")
def route_endpoint(req: RouteRequest):
    p = req.prompt.strip()
    p_lower = p.lower()

    # Быстрый шлюз для очевидных приветствий (экономим GPU время)
    if len(p) < 20 and any(w in p_lower for w in ["привет", "hello", "hi", "здравствуй", "test", "ping"]):
        model_name, comp, rat, tokens = market_catalog.find_optimal_model("greeting", 0.05, len(p))
        return {
            "category": "greeting",
            "complexity": comp,
            "selected_model": model_name,
            "benchmark_rationale": rat,
            "estimated_tokens": tokens,
            "source": "Colab GPU (Live Market Optimizer)"
        }

    messages = [
        {"role": "system", "content": SYSTEM_METRIC_PROMPT},
        {"role": "user", "content": p}
    ]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer([text], return_tensors="pt").to("cuda")

    with torch.no_grad():
        outputs = model.generate(
            **inputs,
            max_new_tokens=100,
            do_sample=False,
            repetition_penalty=1.1,
            pad_token_id=tokenizer.eos_token_id,
        )
    raw = tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)

    category = "routine_backend"
    complexity_score = 0.5
    try:
        s = raw.find("{")
        e = raw.rfind("}")
        parsed = json.loads(raw[s:e+1])
        category = parsed.get("category", category)
        complexity_score = float(parsed.get("complexity_score", 0.5))
    except Exception:
        # Эвристическая оценка, если JSON не распарсился
        if any(w in p_lower for w in ["simd", "avx", "интринсик", "dht", "kademlia"]):
            category = "system_architecture"
            complexity_score = 0.95
        elif any(w in p_lower for w in ["deadlock", "гонк", "race condition", "mpsc", "ffi"]):
            category = "concurrency_debugging"
            complexity_score = 0.85
        elif any(w in p_lower for w in ["vue", "css", "верстк", "ui", "tailwind", "компонент"]):
            category = "frontend_ui"
            complexity_score = 0.65
        else:
            category = "routine_backend"
            complexity_score = 0.40

    # Запускаем расчет Цена / Качество через MarketCatalog
    selected_model, complexity_label, rationale, est_tokens = market_catalog.find_optimal_model(
        category=category,
        complexity_score=complexity_score,
        prompt_len=len(p)
    )

    return {
        "category": category,
        "complexity": complexity_label,
        "selected_model": selected_model,
        "benchmark_rationale": rationale,
        "estimated_tokens": est_tokens,
        "source": "Colab GPU (Live Market Optimizer)"
    }

@app.get("/health")
def health():
    return {
        "status": "ok",
        "market_models_count": len(market_catalog.catalog),
        "adapter_loaded": adapter_path is not None
    }
