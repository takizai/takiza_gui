# ------------------------------------------------------------------------------
# 3. Загрузка мультимодальной базовой модели Qwen2.5-VL-3B в 4-bit + LoRA
# ------------------------------------------------------------------------------
model_id = "Qwen/Qwen2.5-VL-3B-Instruct"

print("\n⏳ Загрузка базовой модели Qwen2.5-VL-3B на GPU в 4-bit...")
bnb_config = BitsAndBytesConfig(
    load_in_4bit=True,
    bnb_4bit_use_double_quant=True,
    bnb_4bit_quant_type="nf4",
    bnb_4bit_compute_dtype=torch.float16,
)

tokenizer = AutoTokenizer.from_pretrained(adapter_path if os.path.exists(adapter_path) else model_id, trust_remote_code=True)
base_model = Qwen2_5_VLForConditionalGeneration.from_pretrained(
    model_id,
    quantization_config=bnb_config,
    device_map="auto",
    dtype=torch.float16,
    trust_remote_code=True,
)

print(f"🔗 Подключение обученного LoRA-адаптера из {adapter_path}...")
model = PeftModel.from_pretrained(base_model, adapter_path)
model.eval()
print("✅ Модель и адаптер успешно инициализированы в памяти GPU!")

# ------------------------------------------------------------------------------
# 4. Инициализация FastAPI API эндпоинта маршрутизации
# ------------------------------------------------------------------------------
app = FastAPI(title="Takiza MoA Intelligent Router")

class RouteRequest(BaseModel):
    prompt: str

SYSTEM_PROMPT = (
    "You are the Takiza MoA Intelligent Router. Analyze the user's task and input files/context. "
    "Classify task complexity and select the most cost-effective and capable model: "
    "ag/gemini-3.7-flash-high, kmc/kimi-for-coding, cc/claude-sonnet-5, cc/claude-opus-5, cx/gpt-5.6-sol, cx/gpt-6-astra. "
    "Output strictly valid JSON with keys: category, complexity, benchmark_rationale, selected_model, estimated_tokens."
)

@app.post("/route")
def route_endpoint(req: RouteRequest):
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": req.prompt}
    ]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    inputs = tokenizer([text], return_tensors="pt").to("cuda")
    
    with torch.no_grad():
        outputs = model.generate(
            **inputs,
            max_new_tokens=120,
            do_sample=False,
            repetition_penalty=1.1,
            pad_token_id=tokenizer.eos_token_id,
        )
    raw = tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
    try:
        s = raw.find("{")
        e = raw.rfind("}")
        return json.loads(raw[s:e+1])
    except Exception:
        return {
            "category": "general",
            "complexity": "medium",
            "benchmark_rationale": "Fallback route from daemon",
            "selected_model": "ag/gemini-3.7-flash-high",
            "estimated_tokens": 500
        }

@app.get("/health")
def health():
    return {"status": "ok", "service": "takiza-moa-router", "gpu": torch.cuda.get_device_name(0)}
